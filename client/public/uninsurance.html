Step 4 — HRSA Integration

Clinic finder in 10 lines

Paste into hrsa.service.ts:

const HRSA = 'https://findahealthcenter.hrsa.gov/api/v1';

export async function findClinics(zip: string) {
  const r = await fetch(
    `${HRSA}/health-centers?zip=${zip}&radius=25`
  );
  return (await r.json()).data;
}
Step 5 — Payment Lanes

InstaMed + Optum routing

// In BookingScreen.tsx
